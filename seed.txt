# MiniDB: схема и тестовые данные (5 datasets, 5 models, 20 experiments)
# Загрузка: ./minidb --reset --script seed.txt

create table datasets (id int, name string, task string, size int) primary key id
create table models (id int, name string, version string) primary key id
create table experiments (id int, dataset_id int, model_id int, accuracy float, status string) primary key id foreign key dataset_id references datasets.id foreign key model_id references models.id
